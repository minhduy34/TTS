\begin{frame}{10. Thiết lập thực nghiệm và độ đo}
\small
\textbf{Dữ liệu:} 20 câu tiếng Việt tự nhiên $\times$ 4 mức độ (Easy, Medium, Hard, Extra Hard), mỗi mức 5 câu
\vspace{0.2cm}
\[
\text{EX}=\frac{1}{N}\sum_{n=1}^{N}\mathbb{I}\big(\text{Exec}(\hat Y_n,D)=\text{Exec}(Y_n,D)\big)
\qquad
\text{EM}=\frac{1}{N}\sum_{n=1}^{N}\prod_{k}\mathbb{I}\big(C_k(\hat Y_n)=C_k(Y_n)\big)
\]
\[
\text{CSR}=\frac{1}{N}\sum_{n=1}^{N}\mathbb{I}\big(\text{không lỗi thực thi}\big)
\]
\begin{itemize}
  \item 3 cấu hình: Baseline $\to$ +Selector $\to$ Full Pipeline
  \item Chạy tự động: \texttt{run\_benchmark.py}
\end{itemize}
\end{frame}
